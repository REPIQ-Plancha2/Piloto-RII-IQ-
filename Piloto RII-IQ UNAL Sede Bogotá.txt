];

export default function App() {
  // Top View Toggle: 'STUDENT' (Public form) vs 'ADMIN' (Private analytics dashboard)
  const [viewMode, setViewMode] = useState('STUDENT');

  // Student Form Navigation (Steps 1 to 4)
  const [currentStep, setCurrentStep] = useState(1);

  // Student Profile Info
  const [studentInfo, setStudentInfo] = useState({
    fullName: '',
    documentId: '',
    email: '',
    semester: '5',
    papa: '4.0'
  });
  const [emailError, setEmailError] = useState('');

  // Selected Subjects State: Array of { subject, prereqStatus, timeSlot, priorityRank (1..5) }
  const [selectedList, setSelectedList] = useState([]);
  const [subjectLimitAlert, setSubjectLimitAlert] = useState(false);

  // Filtering
  const [searchQuery, setSearchQuery] = useState('');
  const [categoryFilter, setCategoryFilter] = useState('ALL');

  // Admin Analytics Copy States
  const [copiedJSON, setCopiedJSON] = useState(false);
  const [formSubmitted, setFormSubmitted] = useState(false);

  // Email Validation
  const validateEmail = (val) => {
    if (!val) {
      setEmailError('El correo institucional es obligatorio.');
      return false;
    }
    const clean = val.trim().toLowerCase();
    if (!clean.endsWith('@unal.edu.co')) {
      setEmailError('Debe ingresar su correo institucional terminado en @unal.edu.co');
      return false;
    }
    setEmailError('');
    return true;
  };

  const handleStudentInfoChange = (field, val) => {
    setStudentInfo(prev => ({ ...prev, [field]: val }));
    if (field === 'email') validateEmail(val);
  };

  // Toggle Subject Selection (Max 5)
  const toggleSubject = (subject) => {
    const isSelected = selectedList.some(item => item.subject.id === subject.id);
    if (isSelected) {
      // Remove and re-index priorities
      const updated = selectedList
        .filter(item => item.subject.id !== subject.id)
        .map((item, idx) => ({ ...item, priorityRank: idx + 1 }));
      setSelectedList(updated);
      setSubjectLimitAlert(false);
    } else {
      if (selectedList.length >= MAX_SELECTIONS) {
        setSubjectLimitAlert(true);
        setTimeout(() => setSubjectLimitAlert(false), 4000);
        return;
      }
      const newItem = {
        subject,
        prereqStatus: 'EN_CURSO_ALTO',
        timeSlot: 'MANANA',
        priorityRank: selectedList.length + 1 // Assign next available priority rank (1..5)
      };
      setSelectedList([...selectedList, newItem]);
    }
  };

  // Update specific field for selected subject
  const updateSubjectField = (subjectId, field, val) => {
    setSelectedList(prev =>
      prev.map(item => {
        if (item.subject.id === subjectId) {
          return { ...item, [field]: val };
        }
        return item;
      })
    );
  };

  // Change priority rank (Swap logic to ensure 1 to 5 order without duplicates)
  const changePriorityRank = (subjectId, newRank) => {
    const targetRank = parseInt(newRank, 10);
    const currentIndex = selectedList.findIndex(item => item.subject.id === subjectId);
    if (currentIndex === -1) return;

    const oldRank = selectedList[currentIndex].priorityRank;
    if (oldRank === targetRank) return;

    // Swap ranks with the item that currently holds targetRank
    const updated = selectedList.map(item => {
      if (item.subject.id === subjectId) {
        return { ...item, priorityRank: targetRank };
      }
      if (item.priorityRank === targetRank) {
        return { ...item, priorityRank: oldRank };
      }
      return item;
    });

    // Sort list by priority rank for clean display
    updated.sort((a, b) => a.priorityRank - b.priorityRank);
    setSelectedList(updated);
  };

  // Total Credits calculation
  const totalCredits = useMemo(() => {
    return selectedList.reduce((sum, item) => sum + item.subject.credits, 0);
  }, [selectedList]);

  // Search & Filtered Subjects
  const filteredSubjects = useMemo(() => {
    return INITIAL_SUBJECTS.filter(sub => {
      const matchesSearch = sub.name.toLowerCase().includes(searchQuery.toLowerCase()) || sub.id.includes(searchQuery);
      const matchesCat = categoryFilter === 'ALL' || sub.category === categoryFilter;
      return matchesSearch && matchesCat;
    });
  }, [searchQuery, categoryFilter]);

  // JSON Payload Generator for Admin Panel
  const generatedJSON = useMemo(() => {
    return JSON.stringify({
      meta: {
        piloto: 'RII-IQ 202X',
        sede: 'UNAL Bogotá',
        departamento: 'Ingeniería Química y Ambiental (DIQA)',
        ventanaMedicion: '07-Dic al 16-Dic',
        fechaGeneracion: new Date().toISOString()
      },
      estudiante: studentInfo,
      intenciones: selectedList.map(item => ({
        prioridad: item.priorityRank,
        codigo: item.subject.id,
        nombre: item.subject.name,
        creditos: item.subject.credits,
        categoria: item.subject.category,
        estadoPrerrequisito: item.prereqStatus,
        franjaHoraria: item.timeSlot
      }))
    }, null, 2);
  }, [studentInfo, selectedList]);

  // CSV Generator for Admin Panel
  const generatedCSV = useMemo(() => {
    const header = `"Prioridad","Documento","Correo","Nombre","Semestre","Codigo_Materia","Nombre_Materia","Creditos","Prerrequisito","Franja"\n`;

      `${item.priorityRank},"${studentInfo.documentId}","${studentInfo.email}","${studentInfo.fullName}",${studentInfo.semester},"${item.subject.id}","${item.subject.name}",${item.subject.credits},"${item.prereqStatus}","${item.timeSlot}"`
    ).join('\n');
    return header + rows;
  }, [studentInfo, selectedList]);

  const handleCopyJSON = () => {
    navigator.clipboard.writeText(generatedJSON);
    setCopiedJSON(true);
    setTimeout(() => setCopiedJSON(false), 2500);
  };

  const handleDownloadCSV = () => {
    const blob = new Blob([generatedCSV], { type: 'text/csv;charset=utf-8;' });
    const url = URL.createObjectURL(blob);
    const link = document.createElement('a');
    link.href = url;
    link.setAttribute('download', `RII_IQ_Consolidado_${studentInfo.documentId || 'demanda'}.csv`);
    document.body.appendChild(link);
    link.click();
    document.body.removeChild(link);
  };

  const isStep1Valid = studentInfo.fullName.trim().length >= 3 &&
                       studentInfo.documentId.trim().length >= 5 &&
                       studentInfo.email.toLowerCase().endsWith('@unal.edu.co');

  return (

      <header className="border-b border-slate-800 bg-slate-900/90 backdrop-blur sticky top-0 z-50 shadow-lg">

          {/* Logo & Department Branding */}
          <div className="flex items-center space-x-3">
            <div className="h-10 w-10 rounded-xl bg-gradient-to-br from-emerald-600 to-emerald-900 flex items-center justify-center border border-emerald-500/50 shadow-md">
              <Building2 className="w-5 h-5 text-emerald-200" />
            </div>
            <div>
              <div className="flex items-center space-x-2">
                <span className="text-[10px] font-bold uppercase tracking-wider text-emerald-400 bg-emerald-950 px-2 py-0.5 rounded border border-emerald-800">
                  UNAL Sede Bogotá
                </span>
                <span className="text-[10px] text-slate-400">Plan 2874</span>
              </div>
              <h1 className="text-base font-bold text-white tracking-tight">
                Piloto RII-IQ <span className="text-slate-400 font-normal text-xs">| Depto. Ing. Química y Ambiental</span>
              </h1>
            </div>
          </div>

          {/* VIEW SWITCHER TOGGLE (Student vs Admin) */}
          <div className="flex items-center bg-slate-950 p-1 rounded-xl border border-slate-800 text-xs">
            <button
              onClick={() => setViewMode('STUDENT')}
              className={`flex items-center space-x-1.5 px-3 py-1.5 rounded-lg font-medium transition-all ${
                viewMode === 'STUDENT'
                  ? 'bg-emerald-600 text-white shadow-sm'
                  : 'text-slate-400 hover:text-slate-200'
              }`}
            >
              <Eye className="w-3.5 h-3.5" />
              <span>Vista Estudiante (Pública)</span>
            </button>

            <button
              onClick={() => setViewMode('ADMIN')}
              className={`flex items-center space-x-1.5 px-3 py-1.5 rounded-lg font-medium transition-all ${
                viewMode === 'ADMIN'
                  ? 'bg-emerald-600 text-white shadow-sm'
                  : 'text-slate-400 hover:text-slate-200'
              }`}
            >
              <Lock className="w-3.5 h-3.5 text-amber-400" />
              <span>Vista Administración (Privada)</span>
            </button>
          </div>

        </div>
      </header>

      {/* ==================================================================== */}
      {/* MODE 1: VISTA ESTUDIANTE (Public Registration Form)                  */}
      {/* ==================================================================== */}
      {viewMode === 'STUDENT' && (
        <div className="flex-1 flex flex-col">
          
          {/* Actuarial Informative Banner */}
          <section className="bg-gradient-to-r from-emerald-950 via-slate-900 to-slate-950 border-b border-emerald-900/40 py-3 px-4">
            <div className="max-w-7xl mx-auto flex flex-col md:flex-row items-start md:items-center justify-between gap-3 text-xs">
              <div className="flex items-start space-x-2.5">
                <ShieldCheck className="w-5 h-5 text-emerald-400 shrink-0 mt-0.5" />
                <div>
                  <p className="text-emerald-200 font-medium">
                    <strong className="text-white">Piloto Actuarial de Planeación DIQA (07 al 16 de Diciembre):</strong>
                  </p>
                  <p className="text-slate-300 text-[11px] leading-relaxed mt-0.5">
                    Este formulario recopila la demanda real con certidumbre de final de semestre para planificar contratación docente y apertura de grupos. <span className="text-emerald-400 underline font-medium">No sustituye la citación formal del SIA</span>.
                  </p>
                </div>
              </div>

              <div className="shrink-0 flex items-center space-x-2 bg-slate-900 border border-slate-800 px-3 py-1 rounded-lg text-slate-300">
                <Calendar className="w-3.5 h-3.5 text-emerald-400" />
                <span>Fechas: <strong>7 al 16 de Dic</strong></span>
              </div>
            </div>
          </section>

          {/* Form Wizard Navigation Bar */}
          <div className="max-w-7xl w-full mx-auto px-4 sm:px-6 pt-6">
            <div className="bg-slate-900/80 border border-slate-800 rounded-2xl p-3 sm:p-4 shadow-lg">
              <div className="grid grid-cols-2 md:grid-cols-4 gap-2 sm:gap-3 text-xs">
                
                {/* Step 1 */}
                <button
                  onClick={() => setCurrentStep(1)}
                  className={`flex items-center space-x-2.5 p-2.5 rounded-xl border text-left transition-all ${
                    currentStep === 1
                      ? 'bg-emerald-950 border-emerald-500 text-white shadow-md'
                      : currentStep > 1
                      ? 'bg-slate-950 border-slate-700 text-emerald-400'
                      : 'bg-slate-950/40 border-slate-800 text-slate-500'
                  }`}
                >
                  <div className={`w-7 h-7 rounded-lg flex items-center justify-center font-bold text-xs shrink-0 ${
                    currentStep === 1 ? 'bg-emerald-500 text-slate-950' : currentStep > 1 ? 'bg-emerald-900 text-emerald-300' : 'bg-slate-800 text-slate-500'
                  }`}>
                    {currentStep > 1 ? <Check className="w-4 h-4" /> : '1'}
                  </div>
                  <div className="overflow-hidden">
                    <p className="text-[9px] uppercase font-bold text-slate-400">Paso 1</p>
                    <p className="font-semibold truncate text-[11px]">Datos Estudiante</p>
                  </div>
                </button>

                {/* Step 2 */}
                <button
                  onClick={() => isStep1Valid && setCurrentStep(2)}
                  disabled={!isStep1Valid}
                  className={`flex items-center space-x-2.5 p-2.5 rounded-xl border text-left transition-all ${
                    currentStep === 2
                      ? 'bg-emerald-950 border-emerald-500 text-white shadow-md'
                      : currentStep > 2
                      ? 'bg-slate-950 border-slate-700 text-emerald-400'
                      : 'bg-slate-950/40 border-slate-800 text-slate-500 opacity-60'
                  }`}
                >
                  <div className={`w-7 h-7 rounded-lg flex items-center justify-center font-bold text-xs shrink-0 ${
                    currentStep === 2 ? 'bg-emerald-500 text-slate-950' : currentStep > 2 ? 'bg-emerald-900 text-emerald-300' : 'bg-slate-800 text-slate-500'
                  }`}>
                    {currentStep > 2 ? <Check className="w-4 h-4" /> : '2'}

                    <p className="text-[9px] uppercase font-bold text-slate-400">Paso 2</p>

                  onClick={() => isStep1Valid && selectedList.length > 0 && setCurrentStep(3)}

                    currentStep === 3

                      : currentStep > 3
                      ? 'bg-slate-950 border-slate-700 text-emerald-400'
                      : 'bg-slate-950/40 border-slate-800 text-slate-500 opacity-60'

                    {currentStep > 3 ? <Check className="w-4 h-4" /> : '3'}
                  </div>
                  <div className="overflow-hidden">
                    <p className="text-[9px] uppercase font-bold text-slate-400">Paso 3</p>
                    <p className="font-semibold truncate text-[11px]">Prioridad (1 a 5) & Horarios</p>
                  </div>
                </button>

                {/* Step 4 */}
                <button
                  onClick={() => isStep1Valid && selectedList.length > 0 && setCurrentStep(4)}
                  disabled={!isStep1Valid || selectedList.length === 0}
                  className={`flex items-center space-x-2.5 p-2.5 rounded-xl border text-left transition-all ${
                    currentStep === 4
                      ? 'bg-emerald-950 border-emerald-500 text-white shadow-md'
                      : 'bg-slate-950/40 border-slate-800 text-slate-500 opacity-60'
                  }`}

                  <div className={`w-7 h-7 rounded-lg flex items-center justify-center font-bold text-xs shrink-0 ${
                    currentStep === 4 ? 'bg-emerald-500 text-slate-950' : 'bg-slate-800 text-slate-500'
                  }`}>
                    4

                    <p className="text-[9px] uppercase font-bold text-slate-400">Paso 4</p>
                    <p className="font-semibold truncate text-[11px]">Confirmación</p>

            {/* ------------ STEP 1: STUDENT PROFILE ------------ */}
            {currentStep === 1 && (
              <div className="bg-slate-900 border border-slate-800 rounded-2xl p-6 md:p-8 shadow-xl max-w-2xl mx-auto space-y-6">
                <div className="flex items-center space-x-3 pb-4 border-b border-slate-800">
                  <div className="p-3 bg-emerald-950 border border-emerald-800 rounded-xl text-emerald-400">
                    <GraduationCap className="w-6 h-6" />

                    <h2 className="text-lg font-bold text-white">Información del Estudiante (Plan 2874)</h2>

                    <label className="block font-semibold uppercase text-slate-300 mb-1.5">
                      Nombre Completo <span className="text-emerald-400">*</span>

                      <User className="w-4 h-4 absolute left-3 top-1/2 -translate-y-1/2 text-slate-500" />

                        type="text"
                        placeholder="Ej. Carlos Eduardo Ramírez Castro"
                        value={studentInfo.fullName}
                        onChange={(e) => handleStudentInfoChange('fullName', e.target.value)}
                        className="w-full bg-slate-950 border border-slate-700 rounded-xl py-2.5 pl-10 pr-4 text-white placeholder-slate-500 focus:outline-none focus:border-emerald-500"

                          type="text"
